<script>
const grid=document.getElementById('wallpaper-grid');
const searchIn=document.getElementById('search-input');
const emptyState=document.getElementById('empty-state');
const wpCount=document.getElementById('wp-count');
const lightbox=document.getElementById('lightbox');
const lbImg=document.getElementById('lb-img');
const lbTitle=document.getElementById('lb-title');
const lbClose=document.getElementById('lb-close');
const lbPrev=document.getElementById('lb-prev');
const lbNext=document.getElementById('lb-next');
const dlBtn=document.getElementById('dl-btn');
const resDetected=document.getElementById('res-detected');
const resChange=document.getElementById('res-change');
let cards=Array.from(grid.querySelectorAll('.wp-card'));
let currentIndex=0;
function updateCount(){const v=cards.filter(c=>c.style.display!=='none').length;wpCount.textContent=v+' wallpaper'+(v!==1?'s':'');emptyState.classList.toggle('visible',v===0);}
updateCount();
searchIn.addEventListener('input',function(){const q=this.value.toLowerCase().trim();cards.forEach(c=>{c.style.display=(!q||(c.dataset.name||'').toLowerCase().includes(q))?'':'none';});updateCount();});
resDetected.textContent=window.screen.width+'×'+window.screen.height;
resChange.addEventListener('click',function(){const w=prompt('Enter your screen width:',window.screen.width);const h=prompt('Enter your screen height:',window.screen.height);if(w&&h)resDetected.textContent=w+'×'+h;});
function getVisible(){return cards.filter(c=>c.style.display!=='none');}
function openLightbox(card){const v=getVisible();currentIndex=v.indexOf(card);loadLightbox(v[currentIndex]);lightbox.classList.add('open');document.body.style.overflow='hidden';}
function loadLightbox(card){const src=card.dataset.full||card.dataset.src;const name=card.dataset.name||'wallpaper';lbImg.src=src;lbImg.alt=name;lbTitle.textContent=name;dlBtn.onclick=()=>triggerDownload(src,name);}
function closeLightbox(){lightbox.classList.remove('open');document.body.style.overflow='';lbImg.src='';}
function navigateLightbox(dir){const v=getVisible();currentIndex=(currentIndex+dir+v.length)%v.length;loadLightbox(v[currentIndex]);}
grid.addEventListener('click',function(e){const card=e.target.closest('.wp-card');if(!card)return;if(e.target.closest('.wp-quick-dl')){e.stopPropagation();triggerDownload(card.dataset.src,card.dataset.name);return;}if(card.dataset.src&&card.dataset.src!=='#')openLightbox(card);});
lbClose.addEventListener('click',closeLightbox);
lbPrev.addEventListener('click',()=>navigateLightbox(-1));
lbNext.addEventListener('click',()=>navigateLightbox(1));
lightbox.addEventListener('click',function(e){if(e.target===lightbox)closeLightbox();});
document.addEventListener('keydown',function(e){if(!lightbox.classList.contains('open'))return;if(e.key==='Escape')closeLightbox();if(e.key==='ArrowLeft')navigateLightbox(-1);if(e.key==='ArrowRight')navigateLightbox(1);});
function triggerDownload(src,name){if(!src||src==='#')return;const a=document.createElement('a');a.href=src;a.download=name+'.png';document.body.appendChild(a);a.click();document.body.removeChild(a);}
</script>
